```python
class Building:  
    __year = None # Инкапусляция поля в питоне говно всё равно можно   
    __city = None # присвоить новое значение только не получится вывести
    def __init__(self, year, city):  
        self.year = year  
        self.city = city  
  
    def get_info(self):  
        print("Год:", self.year, ". Город:", self.city, ".", end=" ")  
  
class School(Building):  # Наследование класса Building
    pupils = 0  
  
    def __init__(self, pupils, year, city):  
        super(School, self).__init__(year, city)  
        self.pupils = pupils  
  
    def get_info(self):  
        super().get_info()  # Полиморфизм метода из super(родителя) класса
        print("ученики:", self.pupils)  
  
  
school = School(200, 2011, "Moscow")  
school.get_info()
```

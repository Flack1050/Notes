import webbrowser  # Библиотека для открытия веб сайтов
  
def validator(func):  # Декоратор
    def wrapper(url):  
        if "." in url:  
            func(url)  
        else:  
            print("Ты долбоёб")  
    return wrapper  
  
@validator  # Использование декоратора
def open_url(url):  
    webbrowser.open(url)  
  
  
open_url("https://github.com/Flack1050/Obsidian")
